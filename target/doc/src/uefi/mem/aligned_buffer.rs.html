3use alloc::alloc::{Layout, LayoutError, alloc, dealloc};
4use core::error::Error;
5use core::ptr::NonNull;
6use core::{fmt, slice};
7
8/// Helper class to maintain the lifetime of a memory region allocated with a non-standard alignment.
9/// Facilitates RAII to properly deallocate when lifetime of the object ends.
10///
11/// Note: This uses the global Rust allocator under the hood.
12#[derive(Debug)]
13pub struct AlignedBuffer {
14    ptr: NonNull<u8>,
15    layout: Layout,
16}
17
18impl AlignedBuffer {
19    /// Allocate a new memory region with the requested len and alignment.
20    ///
21    /// # Panics
22    /// This method panics when the allocation fails (e.g. due to an out of memory situation).
23    pub fn from_size_align(len: usize, alignment: usize) -> Result<Self, LayoutError> {
24        let layout = Layout::from_size_align(len, alignment)?;
25        Ok(Self::from_layout(layout))
26    }
27
28    /// Allocate a new memory region with the requested layout.
29    ///
30    /// # Panics
31    /// This method panics when the allocation fails (e.g. due to an out of memory situation).
32    #[must_use]
33    pub fn from_layout(layout: Layout) -> Self {
34        let ptr = unsafe { alloc(layout) };
35        let ptr = NonNull::new(ptr).expect("Allocation failed");
36        Self { ptr, layout }
37    }
38
39    // TODO: Add non-panicking method variants as soon as alloc::AllocError was stabilized (#32838).
40    // - try_from_layout(layout: Layout) -> Result<Self, AllocError>;
41
42    /// Get a pointer to the aligned memory region managed by this instance.
43    #[must_use]
44    pub const fn ptr(&self) -> *const u8 {
45        self.ptr.as_ptr().cast_const()
46    }
47
48    /// Get a mutable pointer to the aligned memory region managed by this instance.
49    #[must_use]
50    pub const fn ptr_mut(&mut self) -> *mut u8 {
51        self.ptr.as_ptr()
52    }
53
54    /// Get the underlying memory region as immutable slice.
55    #[must_use]
56    pub const fn as_slice(&self) -> &[u8] {
57        unsafe { slice::from_raw_parts(self.ptr(), self.size()) }
58    }
59
60    /// Get the underlying memory region as mutable slice.
61    #[must_use]
62    pub const fn as_slice_mut(&mut self) -> &mut [u8] {
63        unsafe { slice::from_raw_parts_mut(self.ptr_mut(), self.size()) }
64    }
65
66    /// Get the size of the aligned memory region managed by this instance.
67    #[must_use]
68    pub const fn size(&self) -> usize {
69        self.layout.size()
70    }
71
72    /// Returns an iterator over the aligned buffer contents.
73    pub fn iter(&self) -> impl Iterator<Item = &u8> {
74        self.as_slice().iter()
75    }
76
77    /// Returns a mutable iterator over the aligned buffer contents.
78    pub fn iter_mut(&mut self) -> impl Iterator<Item = &mut u8> {
79        self.as_slice_mut().iter_mut()
80    }
81
82    /// Fill the aligned memory region with data from the given buffer.
83    ///
84    /// The length of `src` must be the same as `self`.
85    pub fn copy_from_slice(&mut self, src: &[u8]) {
86        assert_eq!(self.size(), src.len());
87        unsafe {
88            self.ptr_mut().copy_from(src.as_ptr(), src.len());
89        }
90    }
91
92    /// Fill the aligned memory region with data from the given iterator.
93    /// If the given iterator is shorter than the buffer, the remaining area will be left untouched.
94    pub fn copy_from_iter(&mut self, src: impl Iterator<Item = u8>) {
95        self.iter_mut()
96            .zip(src)
97            .for_each(|(dst, src_byte)| *dst = src_byte);
98    }
99
100    /// Check the buffer's alignment against the `required_alignment`.
101    pub fn check_alignment(&self, required_alignment: usize) -> Result<(), AlignmentError> {
102        //TODO: use bfr.addr() when it's available
103        if (self.ptr() as usize) % required_alignment != 0 {
104            return Err(AlignmentError); //TODO: use >is_aligned_to< when it's available
105        }
106        Ok(())
107    }
108}
109
110impl Drop for AlignedBuffer {
111    fn drop(&mut self) {
112        unsafe {
113            dealloc(self.ptr_mut(), self.layout);
114        }
115    }
116}
117
118/// The `AlignmentError` is returned if a user-provided buffer doesn't fulfill alignment requirements.
119#[derive(Clone, PartialEq, Eq, Debug)]
120pub struct AlignmentError;
121impl Error for AlignmentError {}
122impl fmt::Display for AlignmentError {
123    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
124        f.write_str("Buffer alignment does not fulfill requirements.")
125    }
126}

128#[cfg(test)]
129mod tests {
130    use super::AlignedBuffer;
131
132    #[test]
133    fn test_invalid_arguments() {
134        // invalid alignments, valid len
135        for request_alignment in [0, 3, 5, 7, 9] {
136            for request_len in [1, 32, 64, 128, 1024] {
137                assert!(AlignedBuffer::from_size_align(request_len, request_alignment).is_err());
138            }
139        }
140    }
141
142    #[test]
143    fn test_allocation_alignment() {
144        for request_alignment in [1, 2, 4, 8, 16, 32, 64, 128] {
145            for request_len in [1_usize, 32, 64, 128, 1024] {
146                let buffer =
147                    AlignedBuffer::from_size_align(request_len, request_alignment).unwrap();
148                assert_eq!(buffer.ptr() as usize % request_alignment, 0);
149                assert_eq!(buffer.size(), request_len);
150            }
151        }
152    }
153
154    #[test]
155    fn test_copy_from_iter() {
156        let src8: [u8; 8] = [1, 2, 3, 4, 5, 6, 7, 8];
157        {
158            // src as large as dst
159            let mut bfr = AlignedBuffer::from_size_align(8, 8).unwrap();
160            bfr.copy_from_iter(src8.iter().cloned());
161            assert_eq!(bfr.as_slice(), src8);
162        }
163        {
164            // src larger than dst
165            let mut bfr = AlignedBuffer::from_size_align(7, 8).unwrap();
166            bfr.copy_from_iter(src8.iter().cloned());
167            assert_eq!(bfr.as_slice(), [1, 2, 3, 4, 5, 6, 7]);
168        }
169        {
170            // src smaller than dst
171            let mut bfr = AlignedBuffer::from_size_align(9, 8).unwrap();
172            bfr.iter_mut().for_each(|dst| *dst = 0); // fill with 0s
173            bfr.copy_from_iter(src8.iter().cloned());
174            assert_eq!(bfr.as_slice(), [1, 2, 3, 4, 5, 6, 7, 8, 0]);
175        }
176    }
