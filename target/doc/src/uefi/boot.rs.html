52/// Global image handle. This is only set by [`set_image_handle`], and it is
53/// only read by [`image_handle`].
54static IMAGE_HANDLE: AtomicPtr<c_void> = AtomicPtr::new(ptr::null_mut());
55
56/// Get the [`Handle`] of the currently-executing image.
57#[must_use]
58pub fn image_handle() -> Handle {
59    let ptr = IMAGE_HANDLE.load(Ordering::Acquire);
60    // Safety: the image handle must be valid. We know it is, because it was set
61    // by `set_image_handle`, which has that same safety requirement.
62    unsafe { Handle::from_ptr(ptr) }.expect("set_image_handle has not been called")
63}

65/// Update the global image [`Handle`].
66///
67/// This is called automatically in the `main` entry point as part of
68/// [`uefi::entry`]. It should not be called at any other point in time, unless
69/// the executable does not use [`uefi::entry`], in which case it should be
70/// called once before calling other boot services functions.

72/// # Safety

77pub unsafe fn set_image_handle(image_handle: Handle) {
78    IMAGE_HANDLE.store(image_handle.as_ptr(), Ordering::Release);
79}

100/// Raises a task's priority level and returns a [`TplGuard`].
101///
102/// The effect of calling `raise_tpl` with a `Tpl` that is below the current
103/// one (which, sadly, cannot be queried) is undefined by the UEFI spec,
104/// which also warns against remaining at high `Tpl`s for a long time.
105///
106/// This function returns an RAII guard that will automatically restore the
107/// original `Tpl` when dropped.
108///
109/// # Safety
110///
111/// Raising a task's priority level can affect other running tasks and
112/// critical processes run by UEFI. The highest priority level is the
113/// most dangerous, since it disables interrupts.
114#[must_use]
115pub unsafe fn raise_tpl(tpl: Tpl) -> TplGuard {
116    let bt = boot_services_raw_panicking();
117    let bt = unsafe { bt.as_ref() };
118
119    TplGuard {
120        old_tpl: unsafe { (bt.raise_tpl)(tpl) },
121    }
122}

124/// Allocates a consecutive set of memory pages using the UEFI allocator.
125///
126/// The buffer will be [`PAGE_SIZE`] aligned. Callers are responsible for
127/// freeing the memory using [`free_pages`].
128///
129/// # Arguments
130/// - `allocation_type`: The [`AllocateType`] to choose the allocation strategy.
131/// - `memory_type`: The [`MemoryType`] used to persist the allocation in the
132///   UEFI memory map. Typically, UEFI OS loaders should allocate memory of
133///   type [`MemoryType::LOADER_DATA`].
134///- `count`: Number of pages to allocate.
135///
136/// # Safety
137///
138/// Using this function is safe, but it returns a raw pointer to uninitialized
139/// memory. The memory must be initialized before creating a reference to it
140/// or reading from it eventually.
141///
142/// # Errors
143///
144/// * [`Status::OUT_OF_RESOURCES`]: allocation failed.
145/// * [`Status::INVALID_PARAMETER`]: `mem_ty` is [`MemoryType::PERSISTENT_MEMORY`],
146///   [`MemoryType::UNACCEPTED`], or in the range <code>[MemoryType::MAX]..=0x6fff_ffff</code>.
147/// * [`Status::NOT_FOUND`]: the requested pages could not be found.
148pub fn allocate_pages(
149    allocation_type: AllocateType,
150    memory_type: MemoryType,
151    count: usize,
152) -> Result<NonNull<u8>> {
153    let bt = boot_services_raw_panicking();
154    let bt = unsafe { bt.as_ref() };
155
156    let (ty, initial_addr) = match allocation_type {
157        AllocateType::AnyPages => (RawAllocateType::ANY_PAGES, 0),
158        AllocateType::MaxAddress(addr) => (RawAllocateType::MAX_ADDRESS, addr),
159        AllocateType::Address(addr) => (RawAllocateType::ADDRESS, addr),
160    };
161
162    let mut addr1 = initial_addr;
163    unsafe { (bt.allocate_pages)(ty, memory_type, count, &mut addr1) }.to_result()?;
164
165    // The UEFI spec allows `allocate_pages` to return a valid allocation at
166    // address zero. Rust does not allow writes through a null pointer (which
167    // Rust defines as address zero), so this is not very useful. Only return
168    // the allocation if the address is non-null.
169    if let Some(ptr) = NonNull::new(addr1 as *mut u8) {
170        return Ok(ptr);
171    }
172
173    // Attempt a second allocation. The first allocation (at address zero) has
174    // not yet been freed, so if this allocation succeeds it should be at a
175    // non-zero address.
176    let mut addr2 = initial_addr;
177    let r = unsafe { (bt.allocate_pages)(ty, memory_type, count, &mut addr2) }.to_result();
178
179    // Free the original allocation (ignoring errors).
180    let _unused = unsafe { (bt.free_pages)(addr1, count) };
181
182    // Return an error if the second allocation failed, or if it is still at
183    // address zero. Otherwise, return a pointer to the second allocation.
184    r?;
185    if let Some(ptr) = NonNull::new(addr2 as *mut u8) {
186        Ok(ptr)
187    } else {
188        Err(Status::OUT_OF_RESOURCES.into())
189    }
190}

194/// # Safety
195///

261/// Queries the `get_memory_map` function of UEFI to retrieve the current
262/// size of the map. Returns a [`MemoryMapMeta`].
263///
264/// It is recommended to add a few more bytes for a subsequent allocation
265/// for the memory map, as the memory map itself also needs heap memory,
266/// and other allocations might occur before that call.
267#[must_use]
268pub(crate) fn memory_map_size() -> MemoryMapMeta {
269    let bt = boot_services_raw_panicking();
270    let bt = unsafe { bt.as_ref() };
271
272    let mut map_size = 0;
273    let mut map_key = MemoryMapKey(0);
274    let mut desc_size = 0;
275    let mut desc_version = 0;
276
277    let status = unsafe {
278        (bt.get_memory_map)(
279            &mut map_size,
280            ptr::null_mut(),
281            &mut map_key.0,
282            &mut desc_size,
283            &mut desc_version,
284        )
285    };
286    assert_eq!(status, Status::BUFFER_TOO_SMALL);
287
288    assert_eq!(
289        map_size % desc_size,
290        0,
291        "Memory map must be a multiple of the reported descriptor size."
292    );
293
294    let mmm = MemoryMapMeta {
295        desc_size,
296        map_size,
297        map_key,
298        desc_version,
299    };
300
301    mmm.assert_sanity_checks();
302
303    mmm
304}

306/// Stores the current UEFI memory map in an UEFI-heap allocated buffer
307/// and returns a [`MemoryMapOwned`].
308///
309/// The implementation tries to mitigate some UEFI pitfalls, such as getting
310/// the right allocation size for the memory map to prevent
311/// [`Status::BUFFER_TOO_SMALL`].
312///
313/// # Arguments
314///
315/// - `mt`: The memory type for the backing memory on the UEFI heap.
316///   Usually, this is [`MemoryType::LOADER_DATA`]. You can also use a
317///   custom type.
318///
319/// # Errors
320///
321/// * [`Status::INVALID_PARAMETER`]: Invalid [`MemoryType`]
322/// * [`Status::OUT_OF_RESOURCES`]: allocation failed.
323///
324/// # Panics
325///
326/// Panics if the memory map can't be retrieved because of
327/// [`Status::BUFFER_TOO_SMALL`]. This behaviour was chosen explicitly as
328/// callers can't do anything about it anyway.
329pub fn memory_map(mt: MemoryType) -> Result<MemoryMapOwned> {
330    let mut buffer = MemoryMapBackingMemory::new(mt)?;
331
332    let meta = get_memory_map(buffer.as_mut_slice());
333
334    if let Err(e) = &meta {
335        // We don't want to confuse users and let them think they should handle
336        // this, as they can't do anything about it anyway.
337        //
338        // This path won't be taken in OOM situations, but only if for unknown
339        // reasons, we failed to properly allocate the memory map.
340        if e.status() == Status::BUFFER_TOO_SMALL {
341            panic!("Failed to get a proper allocation for the memory map");
342        }
343    }
344    let meta = meta?;
345
346    Ok(MemoryMapOwned::from_initialized_mem(buffer, meta))
347}

349/// Calls the underlying `GetMemoryMap` function of UEFI. On success,
350/// the buffer is mutated and contains the map. The map might be shorter
351/// than the buffer, which is reflected by the return value.
352pub(crate) fn get_memory_map(buf: &mut [u8]) -> Result<MemoryMapMeta> {
353    let bt = boot_services_raw_panicking();
354    let bt = unsafe { bt.as_ref() };
355
356    let mut map_size = buf.len();
357    let map_buffer = buf.as_mut_ptr().cast::<MemoryDescriptor>();
358    let mut map_key = MemoryMapKey(0);
359    let mut desc_size = 0;
360    let mut desc_version = 0;
361
362    assert_eq!(
363        (map_buffer as usize) % align_of::<MemoryDescriptor>(),
364        0,
365        "Memory map buffers must be aligned like a MemoryDescriptor"
366    );
367
368    unsafe {
369        (bt.get_memory_map)(
370            &mut map_size,
371            map_buffer,
372            &mut map_key.0,
373            &mut desc_size,
374            &mut desc_version,
375        )
376    }
377    .to_result_with_val(|| MemoryMapMeta {
378        map_size,
379        desc_size,
380        map_key,
381        desc_version,
382    })
383}

385/// Creates an event.
386///
387/// This function creates a new event of the specified type and returns it.
388///
389/// Events are created in a "waiting" state, and may switch to a "signaled"
390/// state. If the event type has flag `NotifySignal` set, this will result in
391/// a callback for the event being immediately enqueued at the `notify_tpl`
392/// priority level. If the event type has flag `NotifyWait`, the notification
393/// will be delivered next time `wait_for_event` or `check_event` is called.
394/// In both cases, a `notify_fn` callback must be specified.
395///
396/// # Safety
397///
398/// This function is unsafe because callbacks must handle exit from boot
399/// services correctly.
400///
401/// # Errors
402///
403/// * [`Status::INVALID_PARAMETER`]: an invalid combination of parameters was provided.
404/// * [`Status::OUT_OF_RESOURCES`]: the event could not be allocated.
405pub unsafe fn create_event(
406    event_ty: EventType,
407    notify_tpl: Tpl,
408    notify_fn: Option<EventNotifyFn>,
409    notify_ctx: Option<NonNull<c_void>>,
410) -> Result<Event> {
411    let bt = boot_services_raw_panicking();
412    let bt = unsafe { bt.as_ref() };
413
414    let mut event = ptr::null_mut();
415
416    // Safety: the argument types of the function pointers are defined
417    // differently, but are compatible and can be safely transmuted.
418    let notify_fn: Option<uefi_raw::table::boot::EventNotifyFn> =
419        unsafe { mem::transmute(notify_fn) };
420
421    let notify_ctx = opt_nonnull_to_ptr(notify_ctx);
422
423    // Now we're ready to call UEFI
424    unsafe { (bt.create_event)(event_ty, notify_tpl, notify_fn, notify_ctx, &mut event) }
425        .to_result_with_val(
426            // OK to unwrap: event is non-null for Status::SUCCESS.
427            || unsafe { Event::from_ptr(event) }.unwrap(),
428        )
429}

431/// Creates an event in an event group.
432///
433/// The event's notification function, context, and task priority are specified
434/// by `notify_fn`, `notify_ctx`, and `notify_tpl`, respectively. The event will
435/// be added to the group of events identified by `event_group`.
436///
437/// If no group is specified by `event_group`, this function behaves as if the
438/// same parameters had been passed to `create_event()`.
439///
440/// Event groups are collections of events identified by a shared GUID where,
441/// when one member event is signaled, all other events are signaled and their
442/// individual notification actions are taken. All events are guaranteed to be
443/// signaled before the first notification action is taken. All notification
444/// functions will be executed in the order specified by their `Tpl`.
445///
446/// An event can only be part of a single event group. An event may be removed
447/// from an event group by calling [`close_event`].
448///
449/// The [`EventType`] of an event uses the same values as `create_event()`, except that
450/// `EventType::SIGNAL_EXIT_BOOT_SERVICES` and `EventType::SIGNAL_VIRTUAL_ADDRESS_CHANGE`
451/// are not valid.
452///
453/// For events of type `NOTIFY_SIGNAL` or `NOTIFY_WAIT`, `notify_fn` must be
454/// `Some` and `notify_tpl` must be a valid task priority level. For other event
455/// types these parameters are ignored.
456///
457/// More than one event of type `EventType::TIMER` may be part of a single event
458/// group. However, there is no mechanism for determining which of the timers
459/// was signaled.
460///
461/// This operation is only supported starting with UEFI 2.0; earlier versions
462/// will fail with [`Status::UNSUPPORTED`].
463///
464/// # Safety
465///
466/// The caller must ensure they are passing a valid `Guid` as `event_group`, if applicable.
467///
468/// # Errors
469///
470/// * [`Status::INVALID_PARAMETER`]: an invalid combination of parameters was provided.
471/// * [`Status::OUT_OF_RESOURCES`]: the event could not be allocated.
472pub unsafe fn create_event_ex(
473    event_type: EventType,
474    notify_tpl: Tpl,
475    notify_fn: Option<EventNotifyFn>,
476    notify_ctx: Option<NonNull<c_void>>,
477    event_group: Option<NonNull<Guid>>,
478) -> Result<Event> {
479    let bt = boot_services_raw_panicking();
480    let bt = unsafe { bt.as_ref() };
481
482    if bt.header.revision < Revision::EFI_2_00 {
483        return Err(Status::UNSUPPORTED.into());
484    }
485
486    let mut event = ptr::null_mut();
487
488    // Safety: the argument types of the function pointers are defined
489    // differently, but are compatible and can be safely transmuted.
490    let notify_fn: Option<uefi_raw::table::boot::EventNotifyFn> =
491        unsafe { mem::transmute(notify_fn) };
492
493    unsafe {
494        (bt.create_event_ex)(
495            event_type,
496            notify_tpl,
497            notify_fn,
498            opt_nonnull_to_ptr(notify_ctx),
499            opt_nonnull_to_ptr(event_group),
500            &mut event,
501        )
502    }
503    .to_result_with_val(
504        // OK to unwrap: event is non-null for Status::SUCCESS.
505        || unsafe { Event::from_ptr(event) }.unwrap(),
506    )
507}

597/// Stops execution until an event is signaled.
598///
599/// This function must be called at priority level [`Tpl::APPLICATION`].
600///
601/// The input [`Event`] slice is repeatedly iterated from first to last until
602/// an event is signaled or an error is detected. The following checks are
603/// performed on each event:
604///
605/// * If an event is of type [`NOTIFY_SIGNAL`], then a
606///   [`Status::INVALID_PARAMETER`] error is returned with the index of the
607///   event that caused the failure.
608/// * If an event is in the signaled state, the signaled state is cleared
609///   and the index of the event that was signaled is returned.
610/// * If an event is not in the signaled state but does have a notification
611///   function, the notification function is queued at the event's
612///   notification task priority level. If the execution of the event's
613///   notification function causes the event to be signaled, then the
614///   signaled state is cleared and the index of the event that was signaled
615///   is returned.
616///
617/// To wait for a specified time, a timer event must be included in `events`.
618///
619/// To check if an event is signaled without waiting, an already signaled
620/// event can be used as the last event in the slice being checked, or the
621/// [`check_event`] interface may be used.
622///
623/// # Errors
624///
625/// * [`Status::INVALID_PARAMETER`]: `events` is empty, or one of the events of
626///   of type [`NOTIFY_SIGNAL`].
627/// * [`Status::UNSUPPORTED`]: the current TPL is not [`Tpl::APPLICATION`].
628///
629/// [`NOTIFY_SIGNAL`]: EventType::NOTIFY_SIGNAL
630pub fn wait_for_event(events: &mut [Event]) -> Result<usize, Option<usize>> {
631    let bt = boot_services_raw_panicking();
632    let bt = unsafe { bt.as_ref() };
633
634    let number_of_events = events.len();

896/// Enumerates all handles installed on the system which match a certain query.
897///
898/// # Errors
899///
900/// * [`Status::NOT_FOUND`]: no matching handles found.
901/// * [`Status::BUFFER_TOO_SMALL`]: the buffer is not large enough. The required
902///   size (in number of handles, not bytes) will be returned in the error data.
903pub fn locate_handle<'buf>(
904    search_ty: SearchType,
905    buffer: &'buf mut [MaybeUninit<Handle>],
906) -> Result<&'buf [Handle], Option<usize>> {
907    let bt = boot_services_raw_panicking();
908    let bt = unsafe { bt.as_ref() };
909
910    // Obtain the needed data from the parameters.
911    let (ty, guid, key) = match search_ty {
912        SearchType::AllHandles => (0, ptr::null(), ptr::null()),
913        SearchType::ByRegisterNotify(registration) => {
914            (1, ptr::null(), registration.0.as_ptr().cast_const())
915        }
916        SearchType::ByProtocol(guid) => (2, ptr::from_ref(guid), ptr::null()),
917    };
918
919    let mut buffer_size = buffer.len() * size_of::<Handle>();
920    let status =
921        unsafe { (bt.locate_handle)(ty, guid, key, &mut buffer_size, buffer.as_mut_ptr().cast()) };
922
923    let num_handles = buffer_size / size_of::<Handle>();
924
925    match status {
926        Status::SUCCESS => {
927            let buffer = &buffer[..num_handles];
928            // SAFETY: the entries up to `num_handles` have been initialized.
929            let handles = unsafe { maybe_uninit_slice_assume_init_ref(buffer) };
930            Ok(handles)
931        }
932        Status::BUFFER_TOO_SMALL => Err(Error::new(status, Some(num_handles))),
933        _ => Err(Error::new(status, None)),
934    }
935}

968/// Returns all the handles implementing a certain [`Protocol`].
969///
970/// # Errors
971///
972/// * [`Status::NOT_FOUND`]: no matching handles.
973#[cfg(feature = "alloc")]
974pub fn find_handles<P: ProtocolPointer + ?Sized>() -> Result<Vec<Handle>> {
975    // Search by protocol.
976    let search_type = SearchType::from_proto::<P>();
977
978    // Determine how much we need to allocate.
979    let num_handles = match locate_handle(search_type, &mut []) {
980        Err(err) => {
981            if err.status() == Status::BUFFER_TOO_SMALL {
982                err.data().expect("error data is missing")
983            } else {
984                return Err(err.to_err_without_payload());
985            }
986        }
987        // This should never happen: if no handles match the search then a
988        // `NOT_FOUND` error should be returned.
989        Ok(_) => panic!("locate_handle should not return success with empty buffer"),
990    };
991
992    // Allocate a large enough buffer without pointless initialization.
993    let mut handles = Vec::with_capacity(num_handles);
994
995    // Perform the search.
996    let num_handles = locate_handle(search_type, handles.spare_capacity_mut())
997        .discard_errdata()?
998        .len();
999
1000    // Mark the returned number of elements as initialized.
1001    unsafe {
1002        handles.set_len(num_handles);
1003    }
1004
1005    // Emit output, with warnings
1006    Ok(handles)
1007}

1047/// Opens a [`Protocol`] interface for a handle.
1048///
1049/// See also [`open_protocol_exclusive`], which provides a safe subset of this
1050/// functionality.
1051///
1052/// This function attempts to get the protocol implementation of a handle, based
1053/// on the [protocol GUID].
1054///
1055/// See [`OpenProtocolParams`] and [`OpenProtocolAttributes`] for details of the
1056/// input parameters.
1057///
1058/// If successful, a [`ScopedProtocol`] is returned that will automatically
1059/// close the protocol interface when dropped.
1060///
1061/// [protocol GUID]: uefi::data_types::Identify::GUID
1062///
1063/// # Safety
1064///
1065/// This function is unsafe because it can be used to open a protocol in ways
1066/// that don't get tracked by the UEFI implementation. This could allow the
1067/// protocol to be removed from a handle, or for the handle to be deleted
1068/// entirely, while a reference to the protocol is still active. The caller is
1069/// responsible for ensuring that the handle and protocol remain valid until the
1070/// `ScopedProtocol` is dropped.
1071///
1072/// # Errors
1073///
1074/// * [`Status::INVALID_PARAMETER`]: an invalid combination of `params` and
1075///   `attributes` was provided.
1076/// * [`Status::UNSUPPORTED`]: the handle does not support the protocol.
1077/// * [`Status::ACCESS_DENIED`] or [`Status::ALREADY_STARTED`]: the protocol is
1078///   already open in a way that is incompatible with the new request.
1079pub unsafe fn open_protocol<P: ProtocolPointer + ?Sized>(
1080    params: OpenProtocolParams,
1081    attributes: OpenProtocolAttributes,
1082) -> Result<ScopedProtocol<P>> {
1083    let bt = boot_services_raw_panicking();
1084    let bt = unsafe { bt.as_ref() };
1085
1086    let mut interface = ptr::null_mut();
1087    unsafe {
1088        (bt.open_protocol)(
1089            params.handle.as_ptr(),
1090            &P::GUID,
1091            &mut interface,
1092            params.agent.as_ptr(),
1093            Handle::opt_to_ptr(params.controller),
1094            attributes as u32,
1095        )
1096    }
1097    .to_result_with_val(|| {
1098        let interface = if interface.is_null() {
1099            None
1100        } else {
1101            NonNull::new(unsafe { P::mut_ptr_from_ffi(interface) })
1102        };
1103        ScopedProtocol {
1104            interface,
1105            open_params: params,
1106        }
1107    })
1108}

1301/// Convenient wrapper to exit UEFI boot services along with corresponding
1302/// essential steps to get the memory map.
1303///
1304/// Exiting UEFI boot services requires a **non-trivial sequence of steps**,
1305/// including safe retrieval and finalization of the memory map. This wrapper
1306/// ensures a safe and spec-compliant transition from UEFI boot services to
1307/// runtime services by retrieving the system memory map and invoking
1308/// `ExitBootServices()` with the correct memory map key, retrying if necessary.
1309///
1310/// After this function completes, UEFI hands over control of the hardware
1311/// to the executing OS loader, which implies that the UEFI boot services
1312/// are shut down and cannot be used anymore. Only UEFI configuration tables
1313/// and runtime services can be used.
1314///
1315/// Since the boot services function to free memory is no longer available after
1316/// this function returns, the allocation will not be freed on drop. It will
1317/// however be reflected by the memory map itself (self-contained).
1318///
1319/// Note that once the boot services are exited, associated loggers and
1320/// allocators can't use the boot services anymore. For the corresponding
1321/// abstractions provided by this crate (see the [`helpers`] module),
1322/// invoking this function will automatically disable them. If the
1323/// `global_allocator` feature is enabled, attempting to use the allocator
1324/// after exiting boot services will panic.
1325///
1326/// # Arguments
1327/// - `custom_memory_type`: The [`MemoryType`] for the UEFI allocation that will
1328///   store the final memory map. If you pass `None`, this defaults to the
1329///   recommended default value of [`MemoryType::LOADER_DATA`]. If you want a
1330///   specific memory region for the memory map, you can pass the desired
1331///   [`MemoryType`].
1332///
1333/// # Safety
1334///
1335/// The caller is responsible for ensuring that no references to
1336/// boot-services data remain. A non-exhaustive list of resources to check:
1337///
1338/// * All protocols will be invalid after exiting boot services. This
1339///   includes the [`Output`] protocols attached to stdout/stderr. The
1340///   caller must ensure that no protocol references remain.
1341/// * The pool allocator is not usable after exiting boot services. Types
1342///   such as [`PoolString`] which call [`free_pool`] on drop
1343///   must be cleaned up before calling `exit_boot_services`, or leaked to
1344///   avoid drop ever being called.
1345/// * All data in the memory map marked as
1346///   [`MemoryType::BOOT_SERVICES_CODE`] and
1347///   [`MemoryType::BOOT_SERVICES_DATA`] will become free memory.
1348///
1349/// # Errors
1350///
1351/// This function will fail if it is unable to allocate memory for
1352/// the memory map, if it fails to retrieve the memory map, or if
1353/// exiting boot services fails (with up to one retry).
1354///
1355/// All errors are treated as unrecoverable because the system is
1356/// now in an undefined state. Rather than returning control to the
1357/// caller, the system will be reset.
1358///
1359/// [`helpers`]: crate::helpers
1360/// [`Output`]: crate::proto::console::text::Output
1361/// [`PoolString`]: crate::data_types::PoolString
1362#[must_use]
1363pub unsafe fn exit_boot_services(custom_memory_type: Option<MemoryType>) -> MemoryMapOwned {
1364    // LOADER_DATA is the default and also used by the Linux kernel:
1365    // https://elixir.bootlin.com/linux/v6.13.7/source/drivers/firmware/efi/libstub/mem.c#L24
1366    let memory_type = custom_memory_type.unwrap_or(MemoryType::LOADER_DATA);
1367    crate::helpers::exit();
1368
1369    let mut buf = MemoryMapBackingMemory::new(memory_type).expect("Failed to allocate memory");
1370
1371    // Calling `exit_boot_services` can fail if the memory map key is not
1372    // current. Retry a second time if that occurs. This matches the
1373    // behavior of the Linux kernel:
1374    // https://github.com/torvalds/linux/blob/e544a0743/drivers/firmware/efi/libstub/efi-stub-helper.c#L375
1375    let mut status = Status::ABORTED;
1376    for _ in 0..2 {
1377        match unsafe { get_memory_map_and_exit_boot_services(buf.as_mut_slice()) } {
1378            Ok(memory_map) => {
1379                return MemoryMapOwned::from_initialized_mem(buf, memory_map);
1380            }
1381            Err(err) => {
1382                log::error!("Error retrieving the memory map for exiting the boot services");
1383                status = err.status()
1384            }
1385        }
1386    }
1387
1388    // Failed to exit boot services.
1389    log::warn!("Resetting the machine");
1390    runtime::reset(ResetType::COLD, status, None);
1391}

1420///
1421/// UEFI will start a 5-minute countdown after an UEFI image is loaded.  The
1422/// image must either successfully load an OS and exit boot services in that
1423/// time, or disable the watchdog.
1424///

1443///   hardware error.

1445    timeout_in_seconds: usize,
1446    watchdog_code: u64,
1447    data: Option<&mut [u16]>,
1448) -> Result {
1449    let bt = boot_services_raw_panicking();

1451

1453        .map(|d| {

1459        })
1460        .unwrap_or((0, ptr::null_mut()));
1461
1462    unsafe { (bt.set_watchdog_timer)(timeout_in_seconds, watchdog_code, data_len, data) }

1464}

1574/// interface on drop.
1575///
1576/// Most protocols have interface data associated with them. `ScopedProtocol`
1577/// implements [`Deref`] and [`DerefMut`] to access this data. A few protocols
1578/// (such as [`DevicePath`] and [`LoadedImageDevicePath`]) may be installed with
1579/// null interface data, in which case [`Deref`] and [`DerefMut`] will
1580/// panic. The [`get`] and [`get_mut`] methods may be used to access the
1581/// optional interface data without panicking.
1582///
1583/// [`DevicePath`]: crate::proto::device_path::DevicePath
1584/// [`LoadedImageDevicePath`]: crate::proto::device_path::LoadedImageDevicePath
1585/// [`get`]: ScopedProtocol::get
1586/// [`get_mut`]: ScopedProtocol::get_mut
1587#[derive(Debug)]
1588pub struct ScopedProtocol<P: Protocol + ?Sized> {
1589    /// The protocol interface.
1590    interface: Option<NonNull<P>>,
1591    open_params: OpenProtocolParams,
1592}
1593
1594impl<P: Protocol + ?Sized> ScopedProtocol<P> {
1595    /// Returns the [`OpenProtocolParams`] used to open the [`ScopedProtocol`].
1596    #[must_use]
1597    pub const fn open_params(&self) -> OpenProtocolParams {
1598        self.open_params
1599    }
1600}

1602impl<P: Protocol + ?Sized> Drop for ScopedProtocol<P> {
1603    fn drop(&mut self) {
1604        let bt = boot_services_raw_panicking();
1605        let bt = unsafe { bt.as_ref() };
1606
1607        let status = unsafe {
1608            (bt.close_protocol)(
1609                self.open_params.handle.as_ptr(),
1610                &P::GUID,
1611                self.open_params.agent.as_ptr(),
1612                Handle::opt_to_ptr(self.open_params.controller),
1613            )
1614        };
1615        // All of the error cases for close_protocol boil down to
1616        // calling it with a different set of parameters than what was
1617        // passed to open_protocol. The public API prevents such errors,
1618        // and the error can't be propagated out of drop anyway, so just
1619        // assert success.
