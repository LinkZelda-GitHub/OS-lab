15/// Error returned by [`CString16::try_from::<&str>`].
16#[derive(Clone, Copy, Debug, Eq, PartialEq)]
17pub enum FromStrError {
18    /// Character conversion error.
19    InvalidChar,
20    /// Nul character found in the input.
21    InteriorNul,
22}
23
24impl Display for FromStrError {
25    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
26        write!(
27            f,
28            "UCS-2 Conversion Error: {}",
29            match self {
30                Self::InvalidChar => "Invalid character",
31                Self::InteriorNul => "Interior null terminator",
32            }
33        )
34    }
35}
36
37impl core::error::Error for FromStrError {}

39/// An owned UCS-2 null-terminated string.
40///
41/// For convenience, a [`CString16`] is comparable with `&str` and `String` from
42/// the standard library through the trait [`EqStrUntilNul`].
43///
44/// # Examples
45///
46/// Round-trip conversion from a [`&str`] to a `CString16` and back:
47///
48/// ```
49/// use uefi::CString16;
50///
51/// let s = CString16::try_from("abc").unwrap();
52/// assert_eq!(s.to_string(), "abc");
53/// ```
54#[derive(Clone, Debug, Eq, PartialEq, Ord, PartialOrd, Hash)]
55pub struct CString16(Vec<Char16>);
56
57impl CString16 {
58    /// Creates a new empty string with a terminating null character.
59    #[must_use]
60    pub fn new() -> Self {
61        Self(vec![NUL_16])
62    }
63
64    /// Inserts a character at the end of the string, right before the null
65    /// character.
66    ///
67    /// # Panics
68    /// Panics if the char is a null character.
69    pub fn push(&mut self, char: Char16) {
70        assert_ne!(char, NUL_16, "Pushing a null-character is illegal");
71        let last_elem = self
72            .0
73            .last_mut()
74            .expect("There should be at least a null character");
75        *last_elem = char;
76        self.0.push(NUL_16);
77    }
78
79    /// Extends the string with the given [`CStr16`]. The null character is
80    /// automatically kept at the end.
81    pub fn push_str(&mut self, str: &CStr16) {
82        str.as_slice()
83            .iter()
84            .copied()
85            .for_each(|char| self.push(char));
86    }
87
88    /// Replaces all chars in the string with the replace value in-place.
89    pub fn replace_char(&mut self, search: Char16, replace: Char16) {
90        assert_ne!(search, NUL_16, "Replacing a null character is illegal");
91        assert_ne!(
92            replace, NUL_16,
93            "Replacing with a null character is illegal"
94        );
95        self.0
96            .as_mut_slice()
97            .iter_mut()
98            .filter(|char| **char == search)
99            .for_each(|char| *char = replace);
100    }
101
102    /// Returns the number of characters without the trailing null character.
103    #[must_use]
104    pub fn num_chars(&self) -> usize {
105        self.0.len() - 1
106    }
107
108    /// Returns if the string is empty. This ignores the null character.
109    #[must_use]
110    pub fn is_empty(&self) -> bool {
111        self.num_chars() == 0
112    }
113}
114
115impl Default for CString16 {
116    fn default() -> Self {
117        Self::new()
118    }
119}

121impl TryFrom<&str> for CString16 {
122    type Error = FromStrError;
123
124    fn try_from(input: &str) -> Result<Self, Self::Error> {
125        // Initially allocate one Char16 for each byte of the input, plus
126        // one for the null character. This should be a good guess for ASCII-ish
127        // input.
128        let mut output = Vec::with_capacity(input.len() + 1);
129
130        // Convert to UTF-16, then convert to UCS-2.
131        for c in input.encode_utf16() {
132            let c = Char16::try_from(c).map_err(|_| FromStrError::InvalidChar)?;
133
134            // Check for interior nul chars.
135            if c == NUL_16 {
136                return Err(FromStrError::InteriorNul);
137            }
138
139            output.push(c);
140        }
141
142        // Add trailing nul.
143        output.push(NUL_16);
144
145        Ok(Self(output))
146    }
147}

149impl TryFrom<Vec<u16>> for CString16 {
150    type Error = FromSliceWithNulError;
151
152    fn try_from(input: Vec<u16>) -> Result<Self, Self::Error> {
153        // Try creating a CStr16 from the input. We throw away the
154        // result if successful, but it takes care of all the necessary
155        // validity checks (valid UCS-2, ends in null, contains no
156        // interior nulls).
157        CStr16::from_u16_with_nul(&input)?;
158
159        // Convert the input vector from `u16` to `Char16`.
160        //
161        // Safety: `Char16` is a transparent struct wrapping `u16`, so
162        // the types are compatible. The pattern used here matches the
163        // example in the docs for `into_raw_parts`.
164        let (ptr, len, cap) = vec_into_raw_parts(input);
165        let rebuilt = unsafe {
166            let ptr = ptr.cast::<Char16>();
167            Vec::from_raw_parts(ptr, len, cap)
168        };
169
170        Ok(Self(rebuilt))
171    }
172}

174impl TryFrom<&UnalignedSlice<'_, u16>> for CString16 {
175    type Error = FromSliceWithNulError;
176
177    fn try_from(input: &UnalignedSlice<u16>) -> Result<Self, Self::Error> {
178        let v = input.to_vec();
179        Self::try_from(v)
180    }
181}

183impl From<&CStr16> for CString16 {

190impl From<&CString16> for String {

342    /// This tests the following UCS-2 string functions:
343    /// - runtime constructor
344    /// - len()
345    /// - push() / push_str()
346    /// - to rust string
347    #[test]
348    fn test_push_str() {
349        let mut str1 = CString16::new();
350        assert_eq!(str1.num_bytes(), 2, "Should have null character");
351        assert_eq!(str1.num_chars(), 0);
352        str1.push(Char16::try_from('h').unwrap());
353        str1.push(Char16::try_from('i').unwrap());
354        assert_eq!(str1.num_chars(), 2);
355
356        let mut str2 = CString16::new();
357        str2.push(Char16::try_from('!').unwrap());
358
359        str2.push_str(str1.as_ref());
360        assert_eq!(str2.num_chars(), 3);
361
362        let rust_str = String::from(&str2);
363        assert_eq!(rust_str, "!hi");
364    }
