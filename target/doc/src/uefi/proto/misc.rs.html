12/// Timestamp [`Protocol`].
13///
14/// Protocol for retrieving a high-resolution timestamp counter.
15///
16/// # Note
17/// If your UEFI firmware not support timestamp protocol which first added at
18/// UEFI spec 2.4 2013. you also could use `RDTSC` in rust, here is a demo
19/// [Slint-UI](https://github.com/slint-ui/slint/blob/2c0ba2bc0f151eba8d1fa17839fa2ac58832ca80/examples/uefi-demo/main.rs#L28-L62)
20/// who use uefi-rs.
21///
22/// [`Protocol`]: uefi::proto::Protocol
23#[derive(Debug)]
24#[repr(transparent)]
25#[unsafe_protocol(TimestampProtocol::GUID)]
26pub struct Timestamp(TimestampProtocol);
27
28impl Timestamp {
29    /// Get the current value of the timestamp counter.
30    #[must_use]
31    pub fn get_timestamp(&self) -> u64 {
32        unsafe { (self.0.get_timestamp)() }
33    }
34
35    /// Get the properties of the timestamp counter.
36    pub fn get_properties(&self) -> Result<TimestampProperties> {
37        let mut properties = TimestampProperties::default();
38        unsafe { (self.0.get_properties)(&mut properties) }.to_result_with_val(|| properties)
39    }
40}

42/// Reset Notification [`Protocol`].
43///
44/// Protocol to register for a notification when ResetSystem is called.
45///
46/// [`Protocol`]: uefi::proto::Protocol
47#[derive(Debug)]
48#[repr(transparent)]
49#[unsafe_protocol(ResetNotificationProtocol::GUID)]
50pub struct ResetNotification(ResetNotificationProtocol);
51
52impl ResetNotification {
53    /// Register a notification function to be called when ResetSystem() is called.
54    ///
55    ///
56    /// # Example
57    ///
58    /// ```rust
59    /// use log::info;
60    /// use uefi::{boot, Handle};
61    /// use uefi::proto::misc::{ResetNotification};
62    /// use uefi_raw::Status;
63    /// use uefi_raw::table::runtime;
64    ///
65    ///
66    /// // value efi_reset_fn is the type of ResetSystemFn, a function pointer
67    /// unsafe extern "efiapi" fn efi_reset_fn(
68    ///         rt: runtime::ResetType,
69    ///         status: Status,
70    ///         data_size: usize,
71    ///         data: *const u8,
72    /// ){
73    ///     info!("Inside the event callback");
74    ///     info!("do what you want");
75    /// }
76    ///
77    /// pub fn test(image: Handle) {
78    ///
79    ///     let mut rn = boot::open_protocol_exclusive::<ResetNotification>(image)
80    ///         .expect("Failed to open Timestamp protocol");
81    ///
82    ///     rn.register_reset_notify(efi_reset_fn)
83    ///         .expect("Failed to register a reset notification function!");
84    /// }
85    /// ```
86    pub fn register_reset_notify(&mut self, reset_function: ResetSystemFn) -> Result {
87        unsafe { (self.0.register_reset_notify)(&mut self.0, reset_function) }.to_result()
88    }
89
90    /// Remove a reset notification function that was previously registered with [`ResetNotification::register_reset_notify`].
91    pub fn unregister_reset_notify(&mut self, reset_function: ResetSystemFn) -> Result {
92        unsafe { (self.0.unregister_reset_notify)(&mut self.0, reset_function) }.to_result()
93    }
94}
