17/// Nvme Pass Thru Protocol Mode structure.
18///
19/// This contains information regarding the specific capabilities and requirements
20/// of the NVMe controller, such as buffer alignment constraints.
21pub type NvmePassThruMode = uefi_raw::protocol::nvme::NvmExpressPassThruMode;
22
23/// Identifier for an NVMe namespace.
24///
25/// Namespace IDs are used to target specific namespaces on an NVMe device for commands.
26pub type NvmeNamespaceId = u32;

28/// NVMe Pass Thru Protocol.
29///
30/// One protocol instance corresponds to one NVMe controller
31/// (which, most of the time, corresponds to one SSD).
32///
33/// This API offers a safe and convenient, yet still low-level interface to NVMe devices.
34/// It is designed as a foundational layer, leaving higher-level abstractions responsible for implementing
35/// richer storage semantics, device-specific commands, and advanced use cases.
36///
37/// # UEFI Spec Description
38/// The `EFI_NVM_EXPRESS_PASS_THRU_PROTOCOL` provides essential functionality for interacting
39/// with NVMe controllers and namespaces. It allows sending NVMe commands to either the
40/// controller itself or specific namespaces within the controller.
41#[derive(Debug)]
42#[repr(transparent)]
43#[unsafe_protocol(NvmExpressPassThruProtocol::GUID)]
44pub struct NvmePassThru(UnsafeCell<NvmExpressPassThruProtocol>);
45
46impl NvmePassThru {
47    /// Retrieves the mode of the NVMe Pass Thru protocol.
48    ///
49    /// # Returns
50    /// An instance of [`NvmePassThruMode`] describing the NVMe controller's capabilities.
51    #[must_use]
52    pub fn mode(&self) -> NvmePassThruMode {
53        let mut mode = unsafe { (*(*self.0.get()).mode).clone() };
54        mode.io_align = mode.io_align.max(1); // 0 and 1 is the same, says UEFI spec
55        mode
56    }
57
58    /// Retrieves the alignment requirements for I/O buffers.
59    ///
60    /// # Returns
61    /// An alignment value (in bytes) that all I/O buffers must adhere to for successful operation.
62    #[must_use]
63    pub fn io_align(&self) -> u32 {
64        self.mode().io_align
65    }
66
67    /// Allocates an I/O buffer with the necessary alignment for this NVMe Controller.
68    ///
69    /// You can alternatively do this yourself using the [`AlignedBuffer`] helper directly.
70    /// The `nvme` api will validate that your buffers have the correct alignment and error
71    /// if they don't.
72    ///
73    /// # Arguments
74    /// - `len`: The size (in bytes) of the buffer to allocate.
75    ///
76    /// # Returns
77    /// [`AlignedBuffer`] containing the allocated memory.
78    ///
79    /// # Errors
80    /// This method can fail due to alignment or memory allocation issues.
81    pub fn alloc_io_buffer(&self, len: usize) -> Result<AlignedBuffer, LayoutError> {
82        AlignedBuffer::from_size_align(len, self.io_align() as usize)
83    }
84
85    /// Iterate over all valid namespaces on this NVMe controller.
86    ///
87    /// This ignores the 0-namespaces, which corresponds to the controller itself.
88    /// The iterator yields [`NvmeNamespace`] instances representing individual namespaces.
89    ///
90    /// # Returns
91    /// A [`NvmeNamespaceIterator`] for iterating through the namespaces.
92    #[must_use]
93    pub const fn iter_namespaces(&self) -> NvmeNamespaceIterator<'_> {
94        NvmeNamespaceIterator {
95            proto: &self.0,
96            prev: 0xFFFFFFFF,
97        }
98    }
99
100    /// Get the controller namespace (id = 0).
101    /// This can be used to send ADMIN commands.
102    ///
103    /// # Returns
104    /// A [`NvmeNamespaceIterator`] for iterating through the namespaces.
105    #[must_use]
106    pub const fn controller(&self) -> NvmeNamespace<'_> {
107        NvmeNamespace {
108            proto: &self.0,
109            namespace_id: 0,
110        }
111    }
112}

114/// Represents one namespace on an NVMe controller.
115///
116/// A namespace is a shard of storage that the controller can be partitioned into.
117/// Typically, consumer devices only have a single namespace where all the data resides (id 1).
118#[derive(Debug)]
119pub struct NvmeNamespace<'a> {
120    proto: &'a UnsafeCell<NvmExpressPassThruProtocol>,
121    namespace_id: NvmeNamespaceId,
122}
123
124impl NvmeNamespace<'_> {
125    /// Retrieves the namespace identifier (NSID) associated with this NVMe namespace.
126    #[must_use]
127    pub const fn namespace_id(&self) -> NvmeNamespaceId {
128        self.namespace_id
129    }
130
131    /// Get the final device path node for this namespace.
132    ///
133    /// For a full [`crate::proto::device_path::DevicePath`] pointing to this namespace on the
134    /// corresponding NVMe controller.
135    pub fn path_node(&self) -> crate::Result<PoolDevicePathNode> {
136        unsafe {
137            let mut path_ptr: *const DevicePathProtocol = ptr::null();
138            ((*self.proto.get()).build_device_path)(
139                self.proto.get(),
140                self.namespace_id,
141                &mut path_ptr,
142            )
143            .to_result()?;
144            NonNull::new(path_ptr.cast_mut())
145                .map(|p| PoolDevicePathNode(PoolAllocation::new(p.cast())))
146                .ok_or_else(|| Status::OUT_OF_RESOURCES.into())
147        }
148    }
149
150    /// Sends an NVM Express command to this namespace (Namespace ID ≥ 1).
151    ///
152    /// # Arguments
153    /// - `req`: The [`NvmeRequest`] containing the command and associated data to send to the namespace.
154    ///
155    /// # Returns
156    /// - [`NvmeResponse`] containing the results of the operation, such as data and status.
157    ///
158    /// # Errors
159    /// - [`Status::BAD_BUFFER_SIZE`] The NVM Express Command Packet was not executed. The number
160    ///   of bytes that could be transferred is returned in `TransferLength`.
161    /// - [`Status::NOT_READY`] The NVM Express Command Packet could not be sent because the controller
162    ///   is not ready. The caller may retry later.
163    /// - [`Status::DEVICE_ERROR`] A device error occurred while attempting to send the NVM Express
164    ///   Command Packet. Additional status information is available in `NvmeCompletion`.
165    /// - [`Status::INVALID_PARAMETER`] The Namespace ID or the contents of the Command Packet are invalid.
166    ///   The NVM Express Command Packet was not sent, and no additional status information is available.
167    /// - [`Status::UNSUPPORTED`] The command described by the NVM Express Command Packet is not supported
168    ///   by the NVM Express controller. The Command Packet was not sent, and no additional status
169    ///   information is available.
170    /// - [`Status::TIMEOUT`] A timeout occurred while executing the NVM Express Command Packet.
171    ///   Additional status information is available in `NvmeCompletion`.
172    pub fn execute_command<'req>(
173        &mut self,
174        mut req: NvmeRequest<'req>,
175    ) -> crate::Result<NvmeResponse<'req>> {
176        let mut completion = NvmExpressCompletion::default();
177        // prepare cmd packet
178        req.cmd.nsid = self.namespace_id;
179        req.packet.nvme_cmd = &req.cmd;
180        req.packet.nvme_completion = &mut completion;
181        unsafe {
182            ((*self.proto.get()).pass_thru)(
183                self.proto.get(),
184                self.namespace_id,
185                &mut req.packet,
186                ptr::null_mut(),
187            )
188            .to_result_with_val(|| NvmeResponse { req, completion })
189        }
190    }
191}

193/// An iterator over the namespaces of an NVMe controller.
194///
195/// The iterator yields [`NvmeNamespace`] instances, each representing one namespace
196/// on the NVMe controller.
197#[derive(Debug)]
198pub struct NvmeNamespaceIterator<'a> {
199    proto: &'a UnsafeCell<NvmExpressPassThruProtocol>,
200    prev: NvmeNamespaceId,
201}
202
203impl<'a> Iterator for NvmeNamespaceIterator<'a> {
204    type Item = NvmeNamespace<'a>;
205
206    fn next(&mut self) -> Option<Self::Item> {
207        let result =
208            unsafe { ((*self.proto.get()).get_next_namespace)(self.proto.get(), &mut self.prev) };
209        match result {
210            Status::SUCCESS => Some(NvmeNamespace {
211                proto: self.proto,
212                namespace_id: self.prev,
213            }),
214            Status::NOT_FOUND => None,
215            _ => panic!("Must not happen according to spec!"),
216        }
217    }
