34/// Information about the protocol and the TPM device.
35///
36/// Layout compatible with the C type `EFI_TCG2_BOOT_SERVICE_CAPABILITY`.
37#[repr(C)]
38#[derive(Clone, Copy, Debug, Eq, PartialEq, Ord, PartialOrd)]
39pub struct BootServiceCapability {
40    size: u8,
41
42    /// Version of the EFI TCG2 protocol.
43    pub structure_version: Version,
44
45    /// Version of the EFI TCG2 protocol.
46    pub protocol_version: Version,
47
48    /// Bitmap of supported hash algorithms.
49    pub hash_algorithm_bitmap: HashAlgorithm,
50
51    /// Event log formats supported by the firmware.
52    pub supported_event_logs: EventLogFormat,
53
54    present_flag: u8,
55
56    /// Maximum size (in bytes) of a command that can be sent to the TPM.
57    pub max_command_size: u16,
58
59    /// Maximum size (in bytes) of a response that can be provided by the TPM.
60    pub max_response_size: u16,
61
62    /// Manufacturer ID.
63    ///
64    /// See the [TCG Vendor ID registry].
65    ///
66    /// [TCG Vendor ID registry]: https://trustedcomputinggroup.org/resource/vendor-id-registry/
67    pub manufacturer_id: u32,
68
69    /// Maximum number of supported PCR banks (hashing algorithms).
70    pub number_of_pcr_banks: u32,

74    ///
75    /// [`hash_algorithm_bitmap`]: Self::hash_algorithm_bitmap
76    pub active_pcr_banks: HashAlgorithm,
77}

108/// Event type passed to [`Tcg::hash_log_extend_event`].
109///
110/// Layout compatible with the C type `EFI_TCG2_EVENT`.
111///
112/// The TPM v1 spec uses a single generic event type for both creating a
113/// new event and reading an event from the log. The v2 spec splits this
114/// into two structs: `EFI_TCG2_EVENT` for creating events, and
115/// `TCG_PCR_EVENT2` for reading events. To help clarify the usage, our
116/// API renames these types to `PcrEventInputs` and `PcrEvent`,
117/// respectively.
118#[derive(Eq, Pointee)]
119#[repr(C, packed)]
120pub struct PcrEventInputs {
121    size: u32,
122    event_header: EventHeader,
123    event: [u8],
124}
125
126impl PcrEventInputs {
127    /// Create a new `PcrEventInputs` using a byte buffer for storage.
128    ///
129    /// # Errors
130    ///
131    /// Returns [`Status::BUFFER_TOO_SMALL`] if the `buffer` is not large
132    /// enough. The required size will be returned in the error data.
133    ///
134    /// Returns [`Status::INVALID_PARAMETER`] if the `event_data` size is too
135    /// large.
136    pub fn new_in_buffer<'buf>(
137        buffer: &'buf mut [u8],
138        pcr_index: PcrIndex,
139        event_type: EventType,
140        event_data: &[u8],
141    ) -> Result<&'buf mut Self, Option<usize>> {
142        let required_size = size_of::<u32>() + size_of::<EventHeader>() + event_data.len();
143
144        if buffer.len() < required_size {
145            return Err(Error::new(Status::BUFFER_TOO_SMALL, Some(required_size)));
146        }
147        let size_field = u32::try_from(required_size)
148            .map_err(|_| Error::new(Status::INVALID_PARAMETER, None))?;
149
150        let mut ptr: *mut u8 = buffer.as_mut_ptr().cast();
151
152        unsafe {
153            ptr_write_unaligned_and_add(&mut ptr, size_field);
154            ptr_write_unaligned_and_add(
155                &mut ptr,
156                EventHeader {
157                    header_size: u32::try_from(size_of::<EventHeader>()).unwrap(),
158                    header_version: 1,
159                    pcr_index: pcr_index.0,
160                    event_type,
161                },
162            );
163            ptr::copy(event_data.as_ptr(), ptr, event_data.len());
164
165            let ptr: *mut Self =
166                ptr_meta::from_raw_parts_mut(buffer.as_mut_ptr().cast(), event_data.len());
167            Ok(&mut *ptr)
168        }
169    }
170
171    /// Create a new `PcrEventInputs` in a [`Box`].
172    ///
173    /// # Errors
174    ///
175    /// Returns [`Status::INVALID_PARAMETER`] if the `event_data` size is too
176    /// large.
177    #[cfg(feature = "alloc")]
178    pub fn new_in_box(
179        pcr_index: PcrIndex,
180        event_type: EventType,
181        event_data: &[u8],
182    ) -> Result<Box<Self>> {
183        make_boxed(|buf| Self::new_in_buffer(buf, pcr_index, event_type, event_data))
184    }
185}

212#[repr(C, packed)]
213#[derive(Clone, Copy, Debug, PartialEq, Eq)]
214struct AlgorithmDigestSize {
215    algorithm_id: AlgorithmId,
216    digest_size: u16,
217}
218
219#[derive(Clone, Debug)]
220struct AlgorithmDigestSizes<'a>(UnalignedSlice<'a, AlgorithmDigestSize>);
221
222impl AlgorithmDigestSizes<'_> {
223    fn get_size(&self, alg: AlgorithmId) -> Option<u16> {
224        self.0.iter().find_map(|elem| {
225            if { elem.algorithm_id } == alg {
226                Some(elem.digest_size)
227            } else {
228                None
229            }
230        })
231    }
232}

234fn u32_le_from_bytes_at_offset(bytes: &[u8], offset: usize) -> Option<u32> {
235    let bytes = bytes.get(offset..offset + 4)?;
236    // OK to unwrap: we know `bytes` is now of length 4.
237    let val = u32::from_le_bytes(bytes.try_into().unwrap());
238    Some(val)
239}

241/// Header stored at the beginning of the event log.
242///
243/// Layout compatible with the C type `TCG_EfiSpecIDEventStruct`.
244#[derive(Clone, Debug)]
245#[allow(unused)] // We don't current access most of the fields.
246struct EventLogHeader<'a> {
247    platform_class: u32,
248    // major, minor, errata
249    spec_version: (u8, u8, u8),
250    uintn_size: u8,
251    algorithm_digest_sizes: AlgorithmDigestSizes<'a>,
252    vendor_info: &'a [u8],
253    // Size of the whole header event, in bytes.
254    size_in_bytes: usize,
255}
256
257impl<'a> EventLogHeader<'a> {
258    fn new(event: &'a v1::PcrEvent) -> Option<Self> {
259        if event.pcr_index() != PcrIndex(0) {
260            return None;
261        }
262        if { event.event_type() } != EventType::NO_ACTION {
263            return None;
264        }
265        if event.digest() != [0; 20] {
266            return None;
267        }
268
269        let event = &event.event_data();
270        if event.get(..16)? != *b"Spec ID Event03\0" {
271            return None;
272        }
273        let platform_class = u32_le_from_bytes_at_offset(event, 16)?;
274        let version_minor = *event.get(20)?;
275        let version_major = *event.get(21)?;
276        let version_errata = *event.get(22)?;
277        let uintn_size = *event.get(23)?;
278        let number_of_algorithms = usize_from_u32(u32_le_from_bytes_at_offset(event, 24)?);
279        let vendor_info_size_byte_offset =
280            28 + (number_of_algorithms * size_of::<AlgorithmDigestSize>());
281        let vendor_info_size = usize::from(*event.get(vendor_info_size_byte_offset)?);
282
283        // Safety: we know the slice is big enough because we just
284        // safely got the field after the slice (`vendor_info_size`).
285        let algorithm_digest_sizes = unsafe {
286            let ptr: *const AlgorithmDigestSize = event.as_ptr().add(28).cast();
287            AlgorithmDigestSizes(UnalignedSlice::new(ptr, number_of_algorithms))
288        };
289
290        let vendor_info_byte_offset = vendor_info_size_byte_offset + 1;
291        let vendor_info =
292            event.get(vendor_info_byte_offset..vendor_info_byte_offset + vendor_info_size)?;
293
294        // 32 is the size of PcrEventV1 excluding the event data.
295        let size_in_bytes = 32 + vendor_info_byte_offset + vendor_info_size;
296
297        Some(Self {
298            platform_class,
299            spec_version: (version_major, version_minor, version_errata),
300            uintn_size,
301            algorithm_digest_sizes,
302            vendor_info,
303            size_in_bytes,
304        })
305    }
306}

308/// TPM event log as returned by [`Tcg::get_event_log_v2`].
309///
310/// This type of event log can contain multiple hash types (e.g. SHA-1, SHA-256,
311/// SHA-512, etc).
312#[derive(Debug)]
313pub struct EventLog<'a> {
314    // Tie the lifetime to the protocol, and by extension, boot services.
315    _lifetime: PhantomData<&'a Tcg>,
316
317    location: *const u8,
318    last_entry: *const u8,
319
320    is_truncated: bool,
321}
322
323impl EventLog<'_> {
324    /// Iterator of events in the log.
325    #[must_use]
326    pub fn iter(&self) -> EventLogIter<'_> {
327        if let Some(header) = self.header() {
328            // Advance past the header
329            let location = unsafe { self.location.add(header.size_in_bytes) };
330
331            EventLogIter {
332                log: self,
333                location,
334                header: self.header(),
335            }
336        } else {
337            EventLogIter {
338                log: self,
339                location: ptr::null(),
340                header: None,
341            }
342        }
343    }
344
345    /// Header at the beginning of the event log.
346    fn header(&self) -> Option<EventLogHeader<'_>> {
347        // The spec is unclear if the header is present when there are
348        // no entries, so lets assume that `self.location` will be null
349        // if there's no header, and otherwise valid.
350        if self.location.is_null() {
351            None
352        } else {
353            // Safety: we trust that the protocol has given us a valid range
354            // of memory to read from.
355            let event = unsafe { v1::PcrEvent::from_ptr(self.location) };
356            EventLogHeader::new(event)
357        }
358    }
359
360    /// Whether the event log is truncated due to not enough space in the log to
361    /// contain some events.
362    #[must_use]
363    pub const fn is_truncated(&self) -> bool {
364        self.is_truncated
365    }
366}

393/// Iterator over a list of digests.
394#[derive(Debug)]
395pub struct PcrEventDigestIter<'a> {
396    digests: PcrEventDigests<'a>,
397    offset: usize,
398}
399
400impl<'a> Iterator for PcrEventDigestIter<'a> {
401    type Item = (AlgorithmId, &'a [u8]);
402
403    fn next(&mut self) -> Option<Self::Item> {
404        let data = &self.digests.data[self.offset..];
405        let alg = data.get(..2)?;
406        let alg = AlgorithmId(u16::from_le_bytes([alg[0], alg[1]]));
407        let digest_size = usize::from(self.digests.algorithm_digest_sizes.get_size(alg)?);
408        let digest = data.get(2..2 + digest_size)?;
409        self.offset += 2 + digest_size;
410        Some((alg, digest))
411    }
412}

414/// PCR event from an [`EventLog`].
415///
416/// This roughly corresponds to the C type `TCG_PCR_EVENT2`, but is not layout
417/// compatible.
418///
419/// The TPM v1 spec uses a single generic event type for both creating a
420/// new event and reading an event from the log. The v2 spec splits this
421/// into two structs: `EFI_TCG2_EVENT` for creating events, and
422/// `TCG_PCR_EVENT2` for reading events. To help clarify the usage, our
423/// API renames these types to `PcrEventInputs` and `PcrEvent`,
424/// respectively.
425#[derive(Debug)]
426pub struct PcrEvent<'a> {
427    pcr_index: PcrIndex,
428    event_type: EventType,
429    digests: &'a [u8],
430    event_data: &'a [u8],
431
432    // Precalculate the pointer to the next event.
433    next: *const u8,
434
435    // This data from the v2 log header is needed to parse the digest data.
436    algorithm_digest_sizes: AlgorithmDigestSizes<'a>,
437}
438
439impl<'a> PcrEvent<'a> {
440    unsafe fn from_ptr(ptr: *const u8, header: EventLogHeader<'a>) -> Option<Self> {
441        let ptr_u32: *const u32 = ptr.cast();
442        let pcr_index = PcrIndex(unsafe { ptr_u32.read_unaligned() });
443        let event_type = EventType(unsafe { ptr_u32.add(1).read_unaligned() });
444        let digests_count = unsafe { ptr_u32.add(2).read_unaligned() };
445        let digests_ptr: *const u8 = unsafe { ptr.add(12) };
446
447        // Get the byte size of the digests so that the digests iterator
448        // can be safe code.
449        let mut digests_byte_size = 0;
450        let mut elem_ptr = digests_ptr;
451        for _ in 0..digests_count {
452            let algorithm_id = AlgorithmId(unsafe { elem_ptr.cast::<u16>().read_unaligned() });
453            let alg_and_digest_size = size_of::<AlgorithmId>()
454                + usize::from(header.algorithm_digest_sizes.get_size(algorithm_id)?);
455            digests_byte_size += alg_and_digest_size;
456            elem_ptr = unsafe { elem_ptr.add(alg_and_digest_size) };
457        }
458
459        let digests = unsafe { slice::from_raw_parts(digests_ptr, digests_byte_size) };
460        let event_size_ptr = unsafe { digests_ptr.add(digests_byte_size) };
461        let event_size = usize_from_u32(unsafe { event_size_ptr.cast::<u32>().read_unaligned() });
462        let event_data_ptr = unsafe { event_size_ptr.add(4) };
463        let event_data = unsafe { slice::from_raw_parts(event_data_ptr, event_size) };
464
465        Some(Self {
466            pcr_index,
467            event_type,
468            digests,
469            event_data,
470            next: unsafe { event_data_ptr.add(event_size) },
471            algorithm_digest_sizes: header.algorithm_digest_sizes,
472        })
473    }
474
475    /// PCR index for the event.
476    #[must_use]
477    pub const fn pcr_index(&self) -> PcrIndex {
478        self.pcr_index
479    }
480
481    /// Type of event, indicating what type of data is stored in [`event_data`].
482    ///
483    /// [`event_data`]: Self::event_data
484    #[must_use]
485    pub const fn event_type(&self) -> EventType {
486        self.event_type
487    }
488
489    /// Raw event data. The meaning of this data can be determined from
490    /// the [`event_type`].
491    ///
492    /// Note that this data is independent of what is hashed in [`digests`].
493    ///
494    /// [`digests`]: Self::digests
495    /// [`event_type`]: Self::event_type
496    #[must_use]
497    pub const fn event_data(&self) -> &[u8] {
498        self.event_data
499    }
500
501    /// Digests of the data hashed for this event.
502    #[must_use]
503    pub fn digests(&self) -> PcrEventDigests<'_> {
504        PcrEventDigests {
505            data: self.digests,
506            algorithm_digest_sizes: self.algorithm_digest_sizes.clone(),
507        }
508    }
509}

511/// Iterator for events in [`EventLog`].
512#[derive(Debug)]
513pub struct EventLogIter<'a> {
514    log: &'a EventLog<'a>,
515    header: Option<EventLogHeader<'a>>,
516    location: *const u8,
517}
518
519impl<'a> Iterator for EventLogIter<'a> {
520    type Item = PcrEvent<'a>;
521
522    fn next(&mut self) -> Option<Self::Item> {
523        // The spec says that `last_entry` will be null if there are no
524        // events. Presumably `location` will be null as well, but check
525        // both just to be safe.
526        if self.location.is_null() || self.log.last_entry.is_null() {
527            return None;
528        }
529
530        // Safety: we trust that the protocol has given us a valid range
531        // of memory to read from.
532        let event = unsafe { PcrEvent::from_ptr(self.location, self.header.clone()?)? };
533
534        // If this is the last entry, set the location to null so that
535        // future calls to `next()` return `None`.
536        if self.location == self.log.last_entry {
537            self.location = ptr::null();
538        } else {
539            self.location = event.next;
540        }
541
542        Some(event)
543    }
544}

569        let mut location = 0;
570        let mut last_entry = 0;
571        let mut truncated = 0;
572
573        let status = unsafe {
574            (self.0.get_event_log)(
575                &mut self.0,

577                &mut location,
578                &mut last_entry,
579                &mut truncated,

584            let is_truncated = truncated != 0;

587                v1::EventLog::new(location as *const u8, last_entry as *const u8, is_truncated)

646            .to_result()
647        }

661        input_parameter_block: &[u8],
662        output_parameter_block: &mut [u8],
663    ) -> Result {
664        let input_parameter_block_len = u32::try_from(input_parameter_block.len())
665            .map_err(|_| Error::from(Status::BAD_BUFFER_SIZE))?;
666        let output_parameter_block_len = u32::try_from(output_parameter_block.len())
667            .map_err(|_| Error::from(Status::BAD_BUFFER_SIZE))?;
668
669        unsafe {
670            (self.0.submit_command)(
671                &mut self.0,
672                input_parameter_block_len,
673                input_parameter_block.as_ptr(),
674                output_parameter_block_len,
675                output_parameter_block.as_mut_ptr(),
676            )
677            .to_result()
678        }
679    }

681    /// Get a bitmap of the active PCR banks. Each bank corresponds to a hash
682    /// algorithm.
683    pub fn get_active_pcr_banks(&mut self) -> Result<HashAlgorithm> {
684        let mut active_pcr_banks = Tcg2HashAlgorithmBitmap::empty();
685
686        let status = unsafe { (self.0.get_active_pcr_banks)(&mut self.0, &mut active_pcr_banks) };
687
688        status.to_result_with_val(|| HashAlgorithm::from_bits_retain(active_pcr_banks.bits()))
689    }
690
691    /// Set the active PCR banks. Each bank corresponds to a hash
692    /// algorithm. This change will not take effect until the system is
693    /// rebooted twice.
694    pub fn set_active_pcr_banks(&mut self, active_pcr_banks: HashAlgorithm) -> Result {
695        let active_pcr_banks = Tcg2HashAlgorithmBitmap::from_bits_retain(active_pcr_banks.bits());
696        unsafe { (self.0.set_active_pcr_banks)(&mut self.0, active_pcr_banks) }.to_result()
697    }
698
699    /// Get the stored result of calling [`Tcg::set_active_pcr_banks`] in a
700    /// previous boot.
701    ///
702    /// If there was no attempt to set the active PCR banks in a previous boot,
703    /// this returns `None`. Otherwise, it returns a numeric response code:
704    /// * `0x00000000`: Success
705    /// * `0x00000001..=0x00000FFF`: TPM error code
706    /// * `0xfffffff0`: The operation was canceled by the user or timed out
707    /// * `0xfffffff1`: Firmware error
708    pub fn get_result_of_set_active_pcr_banks(&mut self) -> Result<Option<u32>> {
709        let mut operation_present = 0;
710        let mut response = 0;
711
712        let status = unsafe {
713            (self.0.get_result_of_set_active_pcr_banks)(
714                &mut self.0,
715                &mut operation_present,
716                &mut response,
717            )
718        };
719
720        status.to_result_with_val(|| {
721            if operation_present == 0 {
722                None
723            } else {
724                Some(response)
725            }
726        })
727    }
